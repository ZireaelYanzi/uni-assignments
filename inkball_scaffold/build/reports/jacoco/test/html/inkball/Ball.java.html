package inkball;


import processing.core.PApplet;
import processing.core.PImage;
import java.util.Map;
import processing.core.PVector;
import java.util.List;


public class Ball {
   private float x, y; 
   private float vx, vy; 
   private String type; 
   private PImage image;  
   private boolean isRemoved;
   private float radius;

    /**
     * Constructor of the ball class
     */
   public Ball(float x, float y, float vx, float vy, String type, PImage image) {
       this.x = x;
       this.y = y;
       this.vx = vx;
       this.vy = vy;
       this.type = type;
       this.image = image;
       this.isRemoved = false; 
       this.radius = image.width / 2;
   }

   public String getType() {
    return type; 
   }

   public float getVx() {
       return vx;
   }

   public float getX() {
       return x;
   }

   public float getY() {
       return y;
   }

   public float getVy() {
       return vy;
   }

   public PImage getImage() {
    return image;
   }

   public void setVx(float vx) {
       this.vx = vx;
   }


   public void setVy(float vy) {
       this.vy = vy;
   }

   /**
     * It modifies the ball’s behaviour based on interactions with the environment. 
     */
   public void update(char[][] tiles, Map<String, PImage> images, int boardWidth, int boardHeight, List<List<PVector>> lines) {
       float nextX = x + vx;
       float nextY = y + vy;
  
       int nextTileX = (int) (nextX / App.CELLSIZE);
       int nextTileY = (int) ((nextY - App.TOPBAR) / App.CELLSIZE);
  
       int ballWidth = image.width;
       int ballHeight = image.height;
  
       if (nextTileX >= 0 && nextTileX < boardWidth && nextTileY >= 0 && nextTileY < boardHeight) {
           char tile = tiles[nextTileY][nextTileX];
           if (isHole(tile)) {
               isRemoved = true;
               return;
           }
       }
  
       if (nextTileX >= 0 && nextTileX < boardWidth) {
           char tileLeft = tiles[(int) ((y - App.TOPBAR) / App.CELLSIZE)][nextTileX];
           char tileRight = tiles[(int) ((y - App.TOPBAR) / App.CELLSIZE)][(int) ((nextX + ballWidth) / App.CELLSIZE)];
  
           if ((isWall(tileLeft) || isWall(tileRight)) && !isHole(tileLeft)) {
               vx = -vx; 
               nextX = x; 
               changeColor(tileLeft, tileRight, images); 
           }
       }
  
       if (nextTileY >= 0 && nextTileY < boardHeight) {
           char tileTop = tiles[nextTileY][(int) (x / App.CELLSIZE)];
           char tileBottom = tiles[(int) ((nextY + ballHeight - App.TOPBAR) / App.CELLSIZE)][(int) (x / App.CELLSIZE)];
  
           if ((isWall(tileTop) || isWall(tileBottom)) && !isHole(tileTop)) {
               vy = -vy; 
               nextY = y;
               changeColor(tileTop, tileBottom, images); 
           }
       }
  
       for (List<PVector> line : lines) {
           boolean collisionOccurred = false;
           for (int j = 0; j < line.size() - 1; j++) {
               PVector p1 = line.get(j);
               PVector p2 = line.get(j + 1);
  
               if (isCollidingWithLine(p1, p2)) {
                   reflectVelocity(p1, p2);
                   for (PVector point : line) {
                       point.y = App.HEIGHT + 100;
                   }
  
                   collisionOccurred = true; 
                   break;
               }
           }
  
           if (collisionOccurred) {
               break; 
           }
       }
       x = nextX;
       y = nextY;
   }

   private boolean isWall(char tile) {
       return tile == 'X' || tile == '1' || tile == '2' || tile == '3' || tile == '4';
   }

   private boolean isHole(char tile) {
       return tile == 'H';
   }

   protected void changeColor(char tile1, char tile2, Map<String, PImage> images) {
       if (tile1 == '1' || tile2 == '1') {
           image = images.get("B1"); 
       } else if (tile1 == '2' || tile2 == '2') {
           image = images.get("B2"); 
       } else if (tile1 == '3' || tile2 == '3') {
           image = images.get("B3");
       } else if (tile1 == '4' || tile2 == '4') {
           image = images.get("B4"); 
       }
   }

   public boolean isRemoved() {
       return isRemoved;
   }

   public void setRemoved(boolean removed) {
    this.isRemoved = removed; 
    }

   private boolean isCollidingWithLine(PVector p1, PVector p2) {
       float distance = distToSegment(new PVector(x + vx, y + vy), p1, p2);
       return distance <= radius; 
   }

   protected void reflectVelocity(PVector p1, PVector p2) {
       PVector lineDirection = PVector.sub(p2, p1).normalize();
       PVector normal = new PVector(-lineDirection.y, lineDirection.x);
      
       float dotProduct = vx * normal.x + vy * normal.y;
       vx = vx - 2 * dotProduct * normal.x;
       vy = vy - 2 * dotProduct * normal.y;
   }

   /**
     * It calculates the shortest distance from a given point (point)
     * to a line segment defined by two points, v and w, in 2D space. 
     */
   protected float distToSegment(PVector point, PVector v, PVector w) {
       float l2 = v.dist(w) * v.dist(w); 
       if (l2 == 0) return point.dist(v); 
       float t = Math.max(0, Math.min(1, PVector.sub(point, v).dot(PVector.sub(w, v)) / l2));
       PVector projection = PVector.add(v, PVector.mult(PVector.sub(w, v), t));
       return point.dist(projection);
   }


   public void draw(PApplet app) {
       if (!isRemoved) {
           app.image(image, x, y); 
       }
   }
}
