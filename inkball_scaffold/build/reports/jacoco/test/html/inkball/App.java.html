package inkball;


import processing.core.PApplet;
import processing.core.PImage;
import processing.data.JSONArray;
import processing.data.JSONObject;
import processing.event.KeyEvent;
import processing.event.MouseEvent;
import processing.core.PVector;


import java.io.BufferedReader;
import java.io.FileReader;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.HashMap;
import java.util.Random;


/**
* App is the main class for the InkBall game which extends the PApplet class from
* the Processing library to manage graphics rendering and user interaction.
* It handles the game's initialization, drawing, user input, and level management.
*  It also reads game configuration from JSON files and manages balls and the game board.
*/
public class App extends PApplet {


   public static final int CELLSIZE = 32;
   public static final int TOPBAR = 64;
   public static int WIDTH = 576;
   public static int HEIGHT = 640;
   public static final int BOARD_WIDTH = WIDTH / CELLSIZE;
   public static final int BOARD_HEIGHT = 20;


   public static final int FPS = 30;


   public static Random random = new Random();
   public String configPath = "config.json";
   public String img = "src/main/resources/inkball/";
   public String level_path = "level1.txt";


   public Map<String, PImage> images = new HashMap<>();
   public ArrayList<Ball> balls = new ArrayList<>();
   public char[][] tiles = new char[BOARD_HEIGHT][BOARD_WIDTH];
   private List<PVector> linePoints;
   private List<List<PVector>> lines;
   private boolean paused = false;
   protected float[] savedVx, savedVy;
   private int timer;
   private boolean timeUp = false;
   private int startTime;
   private int pausedTime;
   private int remainingTime;
   private int spawnInterval;
   private float spawnTimer;
   private List<String> ballQueue;
   private List<String> remainingBalls;
   private int score = 0;
   private double scoreDecreaseModifier;
   private double scoreIncreaseModifier;
   private JSONObject scoreDecreaseData;
   private JSONObject scoreIncreaseData;
   private int currentLevelIndex = 0;
   private boolean gameEnded = false;
  
   public App() {
       this.configPath = "config.json";
   }


   JSONObject config;


   @Override
   public void settings() {
       size(WIDTH, HEIGHT);
   }


   public int getTimer() {
       return timer;
   }


   public List<String> getRemainingBalls() {
       return remainingBalls;
   }


   public boolean getpaused() {
       return paused;
   }


   public void setPaused(boolean paused) {
       this.paused = paused;
   }
  
   @Override
   public void setup() {
        frameRate(FPS);
        linePoints = new ArrayList<>();
        lines = new ArrayList<>();
        config = loadJSONObject(configPath);
        balls.clear();
        loadGameImages();
    
        loadLevelData(currentLevelIndex); 
        String levelFilePath = "level" + (currentLevelIndex + 1) + ".txt";
        loadLevel(levelFilePath);
    
        score = 0;
        startTime = millis();
}

   /**
    * It loads data from the json file and manages score increment and decrement
    */
   public void loadLevelData(int levelIndex) {
       JSONObject levelConfig = config.getJSONArray("levels").getJSONObject(levelIndex);
       timer = levelConfig.getInt("time");
       spawnInterval = levelConfig.getInt("spawn_interval");
       spawnTimer = spawnInterval;


       scoreIncreaseModifier = levelConfig.getDouble("score_increase_from_hole_capture_modifier");
       scoreDecreaseModifier = levelConfig.getDouble("score_decrease_from_wrong_hole_modifier");


       ballQueue = new ArrayList<>();
       JSONArray ballsArray = levelConfig.getJSONArray("balls");
       for (int i = 0; i < ballsArray.size(); i++) {
           ballQueue.add(ballsArray.getString(i));
       }
       remainingBalls = new ArrayList<>(ballQueue);


       scoreIncreaseData = config.getJSONObject("score_increase_from_hole_capture");
       scoreDecreaseData = config.getJSONObject("score_decrease_from_wrong_hole");


       if (!remainingBalls.isEmpty()) {
           String firstBallType = remainingBalls.remove(0);
           spawnNextBallAtS(firstBallType);
       }
   }


   /**
    * Loads all images used in the game, including tiles, balls, walls, entrypoint, and holes.
    */
   private void loadGameImages() {
       images.put("X", loadImageFromPath("wall0.png"));
       images.put("1", loadImageFromPath("wall1.png"));
       images.put("2", loadImageFromPath("wall2.png"));
       images.put("3", loadImageFromPath("wall3.png"));
       images.put("4", loadImageFromPath("wall4.png"));
       images.put(" ", loadImageFromPath("tile.png"));


       images.put("S", loadImageFromPath("entrypoint.png"));
       images.put("H0", loadImageFromPath("hole0.png"));
       images.put("H1", loadImageFromPath("hole1.png"));
       images.put("H2", loadImageFromPath("hole2.png"));
       images.put("H3", loadImageFromPath("hole3.png"));
       images.put("H4", loadImageFromPath("hole4.png"));


       images.put("B0", loadImageFromPath("ball0.png"));
       images.put("B1", loadImageFromPath("ball1.png"));
       images.put("B2", loadImageFromPath("ball2.png"));
       images.put("B3", loadImageFromPath("ball3.png"));
       images.put("B4", loadImageFromPath("ball4.png"));


       images.put("grey", loadImageFromPath("ball0.png"));
       images.put("orange", loadImageFromPath("ball1.png"));
       images.put("blue", loadImageFromPath("ball2.png"));
       images.put("green", loadImageFromPath("ball3.png"));
       images.put("yellow", loadImageFromPath("ball4.png"));
   }


   public PImage loadImageFromPath(String f) {
       try {
           String path = img + f;
           return super.loadImage(path);
       } catch (Exception e) {
           return null;
       }
   }


   /**
    * read the level file and create the game board based on each line of characters
    */
   protected void loadLevel(String levelFile) {
       try (BufferedReader br = new BufferedReader(new FileReader(levelFile))) {
           String line;
           int row = 0;
           while ((line = br.readLine()) != null && row < BOARD_HEIGHT) {
               for (int col = 0; col < BOARD_WIDTH && col < line.length(); col++) {
                   char tile = line.charAt(col);
                   if (tile == 'B' && col + 1 < line.length()) {
                       char number = line.charAt(col + 1);
                       String key = "" + tile + number;


                       PImage ballImage = images.get(key);
                       if (ballImage != null) {
                           float ballX = col * CELLSIZE;
                           float ballY = row * CELLSIZE + TOPBAR;
                           balls.add(new Ball(ballX, ballY, 2, 2, key, ballImage));
                       }
                       col++;
                   } else {
                       tiles[row][col] = tile;
                   }
               }
               row++;
           }
       } catch (Exception e) {
           e.printStackTrace();
       }
   }
  
   @Override
   public void keyPressed(KeyEvent event) {
       if (event.getKey() == ' ') {
           if (!paused) {
               pauseGame();
           } else {
               resumeGame();
           }
           redraw();
       }
       if (event.getKey() == 'r') {
           setup();
           draw();
       }
   }


   protected void pauseGame() {
       /**
        * save velocity when the game is paused
        */
       paused = true;
       savedVx = new float[balls.size()];
       savedVy = new float[balls.size()];
       for (int i = 0; i < balls.size(); i++) {
           savedVx[i] = balls.get(i).getVx();
           savedVy[i] = balls.get(i).getVy();
           balls.get(i).setVx(0);
           balls.get(i).setVy(0);
       }
       redraw();
       pausedTime = millis();
   }


   private void resumeGame() {
       paused = false;
       for (int i = 0; i < balls.size(); i++) {
           balls.get(i).setVx(savedVx[i]);
           balls.get(i).setVy(savedVy[i]);
       }
       redraw();
       int timePaused = millis() - pausedTime;
       startTime += timePaused;
   }


   @Override
   public void keyReleased() {
   }


   @Override
   public void mousePressed(MouseEvent e) {
       if (mouseButton == LEFT) {
           linePoints.clear();
           linePoints.add(new PVector(mouseX, mouseY));
       }
       if (mouseButton == RIGHT) {
           lines.clear();
           linePoints.clear();
           redraw();
       }
   }


   @Override
   public void mouseDragged(MouseEvent e) {
       if (mouseButton == LEFT) {
           PVector newPoint = new PVector(mouseX, mouseY);
           linePoints.add(newPoint);
       }
   }


   @Override
   public void mouseReleased(MouseEvent e) {
       if (mouseButton == LEFT && linePoints.size() > 1) {
           lines.add(new ArrayList<>(linePoints));
       }
   }
   /**
    * It draws the game board, updates the position of the balls, checks for collisions,
    * draws player-drawn lines, manages the timer, and handles pausing, scoring, and game progression.
    */
   @Override
   public void draw() {
       background(200);


       for (int row = 0; row < BOARD_HEIGHT; row++) {
           for (int col = 0; col < BOARD_WIDTH; col++) {
               image(images.get(" "), col * CELLSIZE, row * CELLSIZE + TOPBAR, CELLSIZE, CELLSIZE);
           }
       }


       for (int row = 0; row < BOARD_HEIGHT; row++) {
           for (int col = 0; col < BOARD_WIDTH; col++) {
               char tile = tiles[row][col];
               if (tile == ' ') continue;


               if (tile == 'H' && col + 1 < BOARD_WIDTH) {
                   char nextChar = tiles[row][col + 1];
                   String key = "" + tile + nextChar;
                   if (images.containsKey(key)) {
                       image(images.get(key), col * CELLSIZE, row * CELLSIZE + TOPBAR, CELLSIZE * 2, CELLSIZE * 2);
                   }
                   col++;
                   continue;
               }


               if (tile == 'B' && col + 1 < BOARD_WIDTH) {
                   char nextChar = tiles[row][col + 1];
                   String key = "" + tile + nextChar;
                   if (images.containsKey(key)) {
                       image(images.get(key), col * CELLSIZE, row * CELLSIZE + TOPBAR);
                   }
                   col++;
                   continue;
               }


               if (images.containsKey(String.valueOf(tile))) {
                   image(images.get(String.valueOf(tile)), col * CELLSIZE, row * CELLSIZE + TOPBAR, CELLSIZE, CELLSIZE);
               }
           }
       }


       if (!paused && !timeUp) {
           balls.removeIf(Ball::isRemoved);
           for (Ball ball : balls) {
               ball.update(tiles, images, BOARD_WIDTH, BOARD_HEIGHT, lines);
               checkBallHoleCollision(ball);
           }
       }


       for (Ball ball : balls) {
           ball.draw(this);
       }


       stroke(0);
       strokeWeight(10);


       for (List<PVector> line : lines) {
           for (int i = 0; i < line.size() - 1; i++) {
               PVector p1 = line.get(i);
               PVector p2 = line.get(i + 1);
               line(p1.x, p1.y, p2.x, p2.y);
           }
       }


       if (linePoints.size() > 1) {
           for (int i = 0; i < linePoints.size() - 1; i++) {
               PVector p1 = linePoints.get(i);
               PVector p2 = linePoints.get(i + 1);
               line(p1.x, p1.y, p2.x, p2.y);
           }
       }


       if (!paused && !timeUp) {
           int elapsedTime = (millis() - startTime) / 1000;
           remainingTime = max(0, timer - elapsedTime);
       }


       fill(0);
       textSize(20);
       text("Score: " + score, WIDTH - 35, TOPBAR - 45);


       textAlign(RIGHT, CENTER);
       textSize(20);
       fill(0);
       text("Time: " + remainingTime, WIDTH - 10, TOPBAR - 20);


       if (remainingTime <= 0 && !timeUp) {
           timeUp = true;
           paused = true;
       }


       if (timeUp) {
           textAlign(CENTER, CENTER);
           textSize(20);
           fill(0);
           text("=== TIME'S UP ===", WIDTH / 2, TOPBAR / 2);
       }


       if (paused && !timeUp) {
           textAlign(CENTER, CENTER);
           textSize(20);
           fill(0);
           text("*** PAUSED ***", WIDTH / 2, TOPBAR / 2);
       }


       fill(0);
       noStroke();
       rect(WIDTH - 570, TOPBAR / 2 - 20, 150, 40);


       if (!remainingBalls.isEmpty()) {
           textAlign(RIGHT, CENTER);
           textSize(20);
           fill(0);
           text(String.format("%.1f", spawnTimer), WIDTH - 370, TOPBAR / 2);


           for (int i = 0; i < min(5, remainingBalls.size()); i++) {
               String ballType = remainingBalls.get(i);
               PImage ballImage = images.get(ballType);
               image(ballImage, WIDTH - (i * (ballImage.width + 5)) - 450, TOPBAR / 2 - ballImage.height / 2);
           }
       }


       if (!paused && !timeUp && !remainingBalls.isEmpty()) {
           spawnTimer -= (1.0 / FPS);
           if (spawnTimer <= 0) {
               String nextBallType = remainingBalls.remove(0);
               spawnNextBallAtS(nextBallType);
               spawnTimer = spawnInterval;
           }
       }


       if (remainingBalls.isEmpty() && balls.isEmpty()) {
           score += remainingTime;
           currentLevelIndex++;
           if (currentLevelIndex == 1) {
               loadLevelData(1); 
               loadLevel("level2.txt"); 
           } else if (currentLevelIndex == 2) {
               loadLevelData(2); 
               loadLevel("level3.txt"); 
           } else {
               gameEnded = true;
               currentLevelIndex = 2;
           }


           remainingTime = timer;
       }


       if (gameEnded) {
           textAlign(CENTER, CENTER);
           textSize(20);
           fill(0);
           text("=== ENDED ===", WIDTH / 2, TOPBAR / 2);
       }
   }


   /**
    * Checks whether a ball has collided with a hole, and updates the score
    * based on whether the ball color matches the hole color.
    */
   public void checkBallHoleCollision(Ball ball) {
       int ballGridX = (int) (ball.getX() / CELLSIZE);
       int ballGridY = (int) ((ball.getY() - TOPBAR) / CELLSIZE);


       if (ballGridY >= 0 && ballGridY < BOARD_HEIGHT && ballGridX >= 0 && ballGridX < BOARD_WIDTH) {
           String holeTile = "" + tiles[ballGridY][ballGridX] + tiles[ballGridY][ballGridX + 1];


           if (holeTile.startsWith("H")) {
               String holeColor = getHoleColor(holeTile);
               String ballType = ball.getType();


               if ((ballType.equals("blue") && holeColor.equals("blue")) ||
                   (ballType.equals("orange") && holeColor.equals("orange")) ||
                   (ballType.equals("green") && holeColor.equals("green")) ||
                   (ballType.equals("yellow") && holeColor.equals("yellow")) ||
                   (ballType.equals("grey") || holeColor.equals("grey"))) {
                   int baseScore = scoreIncreaseData.getInt(ballType);
                   score += baseScore * scoreIncreaseModifier;
               } else {
                   int penalty = scoreDecreaseData.getInt(ballType);
                   score -= penalty * scoreDecreaseModifier;
                   remainingBalls.add(ballType);
               }
               ball.setRemoved(true);
           }
       }
   }


   protected String getHoleColor(String holeTile) {
       if (holeTile.startsWith("H")) {
           char colorCode = holeTile.charAt(1);
           switch (colorCode) {
               case '0': return "grey";
               case '1': return "orange";
               case '2': return "blue";
               case '3': return "green";
               case '4': return "yellow";
               default: return "";
           }
       }
       return "";
   }


   protected void spawnNextBallAtS(String ballType) {
       PImage ballImage = images.get(ballType);
       PVector spawnPoint = findSpawnPoint('S');
       if (spawnPoint != null) {
        float vx = (random.nextBoolean() ? -2 : 2);
        float vy = (random.nextBoolean() ? -2 : 2);
        Ball newBall = new Ball(spawnPoint.x, spawnPoint.y, vx, vy, ballType, ballImage);
        balls.add(newBall);
       }
   }


   protected PVector findSpawnPoint(char spawnChar) {
       for (int row = 0; row < BOARD_HEIGHT; row++) {
           for (int col = 0; col < BOARD_WIDTH; col++) {
               if (tiles[row][col] == spawnChar) {
                   return new PVector(col * CELLSIZE, row * CELLSIZE + TOPBAR);
               }
           }
       }
       return null;
   }

       PApplet.main("inkball.App");
   }
